<script>
(function() {
  var searchInput = document.getElementById('pub-search-input');
  var yearFilter = document.getElementById('pub-year-filter');
  var items = document.querySelectorAll('.pub-item');
  var noResults = document.getElementById('pub-no-results');

  function filter() {
    var query = searchInput.value.toLowerCase().trim();
    var year = yearFilter.value;
    var visibleCount = 0;

    items.forEach(function(item) {
      var title = item.getAttribute('data-title');
      var authors = item.getAttribute('data-authors');
      var itemYear = item.getAttribute('data-year');

      var matchesText = !query ||
        title.indexOf(query) !== -1 ||
        authors.indexOf(query) !== -1 ||
        itemYear.indexOf(query) !== -1;
      var matchesYear = !year || itemYear === year;

      if (matchesText && matchesYear) {
        item.style.display = '';
        visibleCount++;
      } else {
        item.style.display = 'none';
      }
    });

    noResults.style.display = visibleCount === 0 ? '' : 'none';
  }

  searchInput.addEventListener('input', filter);
  yearFilter.addEventListener('change', filter);
})();
</script>
